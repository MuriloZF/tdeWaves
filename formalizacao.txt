Estados = {INICIADOR, OCIOSO, ATIVO, OK}
	S_inicial = {INICIADOR, OCIOSO}
	S_final = {OK}
Restrições = {TR, CN, BL, UI}

Iniciador
	Espontaneamente:
		estado = INICIADOR
		entrada = ()
		aguardando = N(x)
		envia <EXPLORAR> a N(x)

Ocioso
	Recebendo <EXPLORAR> de origem:
		processa <EXPLORAR>
		estado = ATIVO
		entrada = origem
		aguardando = N(x) - {origem}
		envia <EXPLORAR> a N(x) - {origem}
	Se não tiver vizinhos além da origem:
		processa <EXPLORAR>
		envia <ECO> a origem
	Recebendo <ECO> de origem:
		remove origem de aguardando
		envia <ECO> a entrada
	
ATIVO (Iniciador)
	Recebendo <ECO> de origem:
		remove origem de aguardando
		se aguardando = ():
			estado = <OK>
